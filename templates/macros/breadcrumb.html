{# Breadcrumb navigation, ported from KDUI (flask-shadcn-starter). Uses Tabler's own icon font (ti-chevron-right) for the separator instead of an inline SVG, since tabler-icons is already loaded globally. Usage: {% from 'macros/breadcrumb.html' import breadcrumb %} {{ breadcrumb(items=[ {"label": "Home", "url": url_for('dashboard')}, {"label": "Passports", "url": url_for('list_passports')}, {"label": "MP-fcd275d8b0c1", "current": true} ]) }} Parameters: - items: list of {label, url (optional), current (bool, optional)} - aria_label: accessible label for the nav element (default "Breadcrumb") - class_: extra classes for the nav element - attrs: dict of extra attributes for the nav element #} {% macro breadcrumb(items=[], aria_label="Breadcrumb", class_='', attrs={}) %}
    {% for item in items %}
  1. {% if item.current %} {{ item.label }} {% elif item.url %} {{ item.label }} {% else %} {{ item.label }} {% endif %}
    2. {% if not loop.last %}
    3. {% endif %} {% endfor %}
{% endmacro %}
